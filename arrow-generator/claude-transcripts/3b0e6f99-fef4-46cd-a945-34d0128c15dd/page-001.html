    function getValues() {
      return {

    function generateArrow() {
      const v = getValues();
      const centerX = 250;
      const centerY = 150;
      const halfLength = v.length / 2;

      const startX = centerX - halfLength;
      const endX = centerX + halfLength;
      const startY = centerY;
      const endY = centerY;

      let pathD;
      let endAngle = 0;
      let startAngle = Math.PI;

      if (v.curved) {
        const ctrlX = centerX;
        const ctrlY = centerY - v.curveAmount;
        pathD = `M ${startX} ${startY} Q ${ctrlX} ${ctrlY} ${endX} ${endY}`;

        // Calculate tangent angle at end point for curved arrow
        // For quadratic bezier, tangent at t=1 is direction from control to end
        const dx = endX - ctrlX;
        const dy = endY - ctrlY;
        endAngle = Math.atan2(dy, dx);

        // Tangent at t=0 is direction from start to control
        const dxStart = ctrlX - startX;
        const dyStart = ctrlY - startY;
        startAngle = Math.atan2(dyStart, dxStart) + Math.PI;
      } else {
        pathD = `M ${startX} ${startY} L ${endX} ${endY}`;
      }

      // Generate arrow head points
      const headAngleRad = (v.headAngle * Math.PI) / 180;

      function getArrowHeadPoints(tipX, tipY, angle, size) {
        const x1 = tipX - size * Math.cos(angle - headAngleRad);
        const y1 = tipY - size * Math.sin(angle - headAngleRad);
        const x2 = tipX - size * Math.cos(angle + headAngleRad);
        const y2 = tipY - size * Math.sin(angle + headAngleRad);
        return { x1, y1, x2, y2 };
      }

      // End arrow head (pointing right)
      const endHead = getArrowHeadPoints(endX, endY, endAngle, v.headSize);

      // Start arrow head (pointing left) - for double ended
      const startHead = getArrowHeadPoints(startX, startY, startAngle, v.headSize);

      let svgContent = '';

      // Main line/curve
      svgContent += `<path d="${pathD}" fill="none" stroke="${v.color}" stroke-width="${v.strokeWidth}" stroke-linecap="round"/>`;

      // End arrow head
      if (v.filledHeads) {
        svgContent += `<polygon points="${endX},${endY} ${endHead.x1},${endHead.y1} ${endHead.x2},${endHead.y2}" fill="${v.color}"/>`;
      } else {
        svgContent += `<polyline points="${endHead.x1},${endHead.y1} ${endX},${endY} ${endHead.x2},${endHead.y2}" fill="none" stroke="${v.color}" stroke-width="${v.strokeWidth}" stroke-linecap="round" stroke-linejoin="round"/>`;
      }

      // Start arrow head (if double ended)
      if (v.doubleEnded) {
        if (v.filledHeads) {
          svgContent += `<polygon points="${startX},${startY} ${startHead.x1},${startHead.y1} ${startHead.x2},${startHead.y2}" fill="${v.color}"/>`;
        } else {
          svgContent += `<polyline points="${startHead.x1},${startHead.y1} ${startX},${startY} ${startHead.x2},${startHead.y2}" fill="none" stroke="${v.color}" stroke-width="${v.strokeWidth}" stroke-linecap="round" stroke-linejoin="round"/>`;
        }
      }

      // Apply rotation
      arrowGroup.innerHTML = svgContent;
      arrowGroup.setAttribute('transform', `rotate(${v.rotation} ${centerX} ${centerY})`);

      // Update SVG code display
      updateSvgCode();
    }

    function updateSvgCode() {
      const svg = document.getElementById('arrowSvg');
      const svgClone = svg.cloneNode(true);
      svgClone.removeAttribute('id');

      // Clean up the output
      const code = svgClone.outerHTML
        .replace(/\s+/g, ' ')
        .replace(/> </g, '>\n<')
        .trim();

      svgCodeTextarea.value = code;
    }

    function downloadSvg() {
      const svg = document.getElementById('arrowSvg');
      const svgData = svg.outerHTML;
      const blob = new Blob([svgData], { type: 'image/svg+xml' });
      const url = URL.createObjectURL(blob);

      const a = document.createElement('a');
      a.href = url;
      a.download = 'arrow.svg';
      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
      URL.revokeObjectURL(url);
    }

    function copySvgCode() {
      svgCodeTextarea.select();
      navigator.clipboard.writeText(svgCodeTextarea.value).then(() => {
        const btn = event.target;
        const originalText = btn.textContent;
        btn.textContent = 'Copied!';
        setTimeout(() => {
          btn.textContent = originalText;
        }, 1500);
      });
    }

    function resetDefaults() {
      controls.length.value = 200;
      controls.strokeWidth.value = 4;

        }

        generateArrow();
      });
    });

    // Initial render
    generateArrow();
