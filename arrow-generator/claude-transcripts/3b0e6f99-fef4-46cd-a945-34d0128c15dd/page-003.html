    async function copyImage() {
      const svg = document.getElementById('arrowSvg');
      const arrowGroup = document.getElementById('arrowGroup');

      // Get the bounding box of the arrow content
      const bbox = arrowGroup.getBBox();
      const padding = 10;
      const scale = 2; // For higher resolution

      // Create a new SVG with viewBox set to the arrow bounds
      const croppedSvg = svg.cloneNode(true);
      croppedSvg.setAttribute('viewBox', `${bbox.x - padding} ${bbox.y - padding} ${bbox.width + padding * 2} ${bbox.height + padding * 2}`);
      croppedSvg.setAttribute('width', (bbox.width + padding * 2) * scale);
      croppedSvg.setAttribute('height', (bbox.height + padding * 2) * scale);

      const svgData = new XMLSerializer().serializeToString(croppedSvg);
      const svgBlob = new Blob([svgData], { type: 'image/svg+xml;charset=utf-8' });
      const url = URL.createObjectURL(svgBlob);

      const img = new Image();
      img.onload = async () => {
        const canvas = document.createElement('canvas');
        canvas.width = (bbox.width + padding * 2) * scale;
        canvas.height = (bbox.height + padding * 2) * scale;
        const ctx = canvas.getContext('2d');

        // Transparent background
