        const ROWS = 6;
        const COLS = 7;
        let board = [];
        let currentPlayer = 'red';
        let gameOver = false;
        let hoveredCol = -1;

        function initBoard() {
            board = Array(ROWS).fill(null).map(() => Array(COLS).fill(null));
            currentPlayer = 'red';
            gameOver = false;
            hoveredCol = -1;
            render();
            updateStatus();
        }

        function render() {
            const boardEl = document.getElementById('board');
            boardEl.innerHTML = '';

            for (let r = 0; r < ROWS; r++) {
                const rowEl = document.createElement('div');
                rowEl.className = 'row';

                for (let c = 0; c < COLS; c++) {
                    const cell = document.createElement('div');
                    cell.className = 'cell';
                    cell.dataset.row = r;
                    cell.dataset.col = c;

                    if (board[r][c]) {
                        cell.classList.add(board[r][c]);
                    } else if (c === hoveredCol && !gameOver) {
                        cell.classList.add('highlight');
                    }

                    cell.addEventListener('click', () => handleClick(c));
                    cell.addEventListener('mouseenter', () => handleHover(c));
                    cell.addEventListener('mouseleave', () => handleHover(-1));

                    rowEl.appendChild(cell);
                }
                boardEl.appendChild(rowEl);
            }
        }

        function handleHover(col) {
            if (gameOver) return;
            hoveredCol = col;
            render();
        }

        function handleClick(col) {
            if (gameOver) return;

            // Find the lowest empty row in this column
            let row = -1;
            for (let r = ROWS - 1; r >= 0; r--) {
                if (!board[r][col]) {
                    row = r;
                    break;
                }
            }

            if (row === -1) return; // Column is full

            board[row][col] = currentPlayer;

            const winner = checkWinner(row, col);
            if (winner) {
                gameOver = true;
                render();
                highlightWinningCells(winner);
                updateStatus(currentPlayer + ' wins!');
                return;
            }

            if (isBoardFull()) {
                gameOver = true;
                updateStatus("It's a draw!");
                render();
                return;
            }

            currentPlayer = currentPlayer === 'red' ? 'yellow' : 'red';
            updateStatus();
            render();
        }

        function updateStatus(message) {
            const statusEl = document.getElementById('status');
            if (message) {
                const colorClass = message.includes('red') ? 'red' : message.includes('yellow') ? 'yellow' : '';
                statusEl.innerHTML = `<span class="${colorClass}">${message.charAt(0).toUpperCase() + message.slice(1)}</span>`;
            } else {
                statusEl.innerHTML = `Current Player: <span class="${currentPlayer}">${currentPlayer.charAt(0).toUpperCase() + currentPlayer.slice(1)}</span>`;
            }
        }

        function checkWinner(row, col) {
            const directions = [
                [[0, 1], [0, -1]],   // Horizontal
                [[1, 0], [-1, 0]],   // Vertical
                [[1, 1], [-1, -1]], // Diagonal \
                [[1, -1], [-1, 1]]  // Diagonal /
            ];

            for (const [dir1, dir2] of directions) {
                const cells = [[row, col]];

                // Check in first direction
                let r = row + dir1[0];
                let c = col + dir1[1];
                while (r >= 0 && r < ROWS && c >= 0 && c < COLS && board[r][c] === currentPlayer) {
                    cells.push([r, c]);
                    r += dir1[0];
                    c += dir1[1];
                }

                // Check in opposite direction
                r = row + dir2[0];
                c = col + dir2[1];
                while (r >= 0 && r < ROWS && c >= 0 && c < COLS && board[r][c] === currentPlayer) {
                    cells.push([r, c]);
                    r += dir2[0];
                    c += dir2[1];
                }

                if (cells.length >= 4) {
                    return cells;
                }
            }

            return null;
        }

        function highlightWinningCells(cells) {
            for (const [r, c] of cells) {
                const cell = document.querySelector(`.cell[data-row="${r}"][data-col="${c}"]`);
                if (cell) cell.classList.add('winning');
            }
        }

        function isBoardFull() {
            return board[0].every(cell => cell !== null);
        }

        document.getElementById('restart').addEventListener('click', initBoard);

        initBoard();
    </script>

This causes a problem because when you click, the mouseenter event fires first (or during the click), which triggers handleHover, which triggers render(), which rebuilds the DOM - and by the time the click event would fire, the element you clicked on no longer exists.

The fix is to not re-render on hover. Instead, I should update the highlight state without rebuilding the entire DOM. Or better yet, use CSS :hover for the highlight effect and only call render() when the board state actually changes.
