   671→        function calculateScore(category, diceValues) {
   672→            const counts = [0, 0, 0, 0, 0, 0];
   673→            let sum = 0;
   674→
   675→            for (const val of diceValues) {
   676→                counts[val - 1]++;
   677→                sum += val;
   678→            }
   679→
   680→            switch (category) {
   681→                case 'ones': return counts[0] * 1;
   682→                case 'twos': return counts[1] * 2;
   683→                case 'threes': return counts[2] * 3;
   684→                case 'fours': return counts[3] * 4;
   685→                case 'fives': return counts[4] * 5;
   686→                case 'sixes': return counts[5] * 6;
   687→
   688→                case 'threeOfKind':
   689→                    return counts.some(c => c >= 3) ? sum : 0;
   690→
   691→                case 'fourOfKind':
   692→                    return counts.some(c => c >= 4) ? sum : 0;
   693→
   694→                case 'fullHouse':
   695→                    const hasThree = counts.some(c => c === 3);
   696→                    const hasTwo = counts.some(c => c === 2);
   697→                    return (hasThree && hasTwo) ? 25 : 0;
   698→
   699→                case 'smallStraight':
   700→                    const sorted = [...new Set(diceValues)].sort();
   701→                    const str = sorted.join('');
   702→                    return (str.includes('1234') || str.includes('2345') || str.includes('3456')) ? 30 : 0;
   703→
   704→                case 'largeStraight':
   705→                    const sortedL = [...diceValues].sort().join('');
   706→                    return (sortedL === '12345' || sortedL === '23456') ? 40 : 0;
   707→
   708→                case 'yahtzee':
   709→                    return counts.some(c => c === 5) ? 50 : 0;
   710→
   711→                case 'chance':
   712→                    return sum;
   713→
   714→                default:
   715→                    return 0;
   716→            }

                case 'twos': return counts[1] * 2;
                case 'threes': return counts[2] * 3;
                case 'fours': return counts[3] * 4;
                case 'fives': return counts[4] * 5;
                case 'sixes': return counts[5] * 6;

                case 'threeOfKind':
                    return counts.some(c => c >= 3) ? sum : 0;

                case 'fourOfKind':
                    return counts.some(c => c >= 4) ? sum : 0;

                case 'fullHouse':
                    const hasThree = counts.some(c => c === 3);
                    const hasTwo = counts.some(c => c === 2);
                    return (hasThree && hasTwo) ? 25 : 0;

                    return (sortedL === '12345' || sortedL === '23456') ? 40 : 0;

                case 'yahtzee':
                    return counts.some(c => c === 5) ? 50 : 0;

                case 'chance':

                    return 0;
