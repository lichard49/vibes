        function calculateScore(category, diceValues) {
            const counts = [0, 0, 0, 0, 0, 0];

            for (const val of diceValues) {
                counts[val - 1]++;
                sum += val;
            }

            switch (category) {
                case 'ones': return counts[0] * 1;
                case 'twos': return counts[1] * 2;
                case 'threes': return counts[2] * 3;
                case 'fours': return counts[3] * 4;
                case 'fives': return counts[4] * 5;
                case 'sixes': return counts[5] * 6;
                case 'threeOfKind': return counts.some(c => c >= 3) ? sum : 0;
                case 'fourOfKind': return counts.some(c => c >= 4) ? sum : 0;
                case 'fullHouse':
                    return (counts.some(c => c === 3) && counts.some(c => c === 2)) ? 25 : 0;
                case 'smallStraight':
                    const sorted = [...new Set(diceValues)].sort();
                    const str = sorted.join('');
                    return (str.includes('1234') || str.includes('2345') || str.includes('3456')) ? 30 : 0;
                case 'largeStraight':
                    const sortedL = [...diceValues].sort().join('');
                    return (sortedL === '12345' || sortedL === '23456') ? 40 : 0;
                case 'yahtzee': return counts.some(c => c === 5) ? 50 : 0;
                case 'chance': return sum;
                default: return 0;
